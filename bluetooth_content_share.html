<div class="product-container">
<img class="product-image" src="product-thumbnail.jpg" alt="Product Thumbnail">
<div class="image-modal hidden">
<img class="preview-image" src="" alt="Product Preview">
<button class="close">&times;</button>
</div>
</div>


.product-container {
    position: relative;
    display: inline-block;
}

.image-modal {
    position: fixed;
    top: 50%;
    left: 50%;
    transform: translate(-50%, -50%);
    width: 80%;
    max-width: 600px;
    background-color: rgba(0, 0, 0, 0.5);
    padding: 10px;
    display: flex;
    justify-content: center;
    align-items: center;
}

.close {
    margin: 0 5px;
    font-size: 1.5em;
    font-weight: bold;
    cursor: pointer;
}

.hidden {
    display: none;
}

document.addEventListener('DOMContentLoaded', () => {
    const modal = document.querySelector('.image-modal');
    const closeButton = modal.querySelector('.close');
    const productImages = document.querySelectorAll('.product-image');

    productImages.forEach(image => {
        image.addEventListener('click', () => {
            modal.classList.remove('hidden');
            modal.querySelector('.preview-image').setAttribute('src', image.getAttribute('src'));
        });
    });

    closeButton.addEventListener('click', () => {
        modal.classList.add('hidden');
    });

    document.addEventListener('click', event => {
        if (event.target !== modal && event.target !== closeButton && !modal.contains(event.target)) {
            modal.classList.add('hidden');
        }
    });
});
